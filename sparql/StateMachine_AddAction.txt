PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> 
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#> 
PREFIX owl: <http://www.w3.org/2002/07/owl#>
PREFIX factory: <https://w3id.org/ontoeng/factory#>
PREFIX osph: <https://w3id.org/ontoeng/osph#>
PREFIX fsm: <http://www.learninglab.de/~dolog/fsm/fsm.owl#>
PREFIX ifcext: <https://w3id.org/ontoeng/IFC4_ADD1_extension#>
PREFIX cs: <https://w3id.org/ontoeng/controlSystem#>
PREFIX ex: <https://w3id.org/ontoeng/expression#>
PREFIX express: <https://w3id.org/express#> 

DELETE{
	# GRAPH ?g0{?actParOLD ?propOLD ?objOLD .}
	GRAPH ?g0{?actParOLD osph:hasParamValue ?actParValDEL .}
	GRAPH ?g0{?actParOLD osph:hasParamSimpleValue ?actParSimpValDEL .}
	GRAPH ?g0{?actParOLD osph:hasParamVariable ?actParVarDEL .}
	GRAPH ?g0{?actParOLD osph:hasEventPayloadVariable ?actParEvVarDEL .}
	GRAPH ?g0{?actParOLD osph:hasEventPayloadValue ?actParEvValDEL .}	
}
INSERT { 
	GRAPH ?graph { 
		?statemach fsm:contains ?action .
		
		?stateEntry fsm:Entry ?action .
		?stateExit fsm:Exit ?action .	
		?transition osph:hasTransitionAction ?action .	

		?action osph:hasActionParameter ?actPar .		
		# ?actPar a owl:NamedIndividual , osph:ActionParameter .

		?actPar osph:hasParamName ?actParName .
		?actPar osph:hasParamValue ?actParValNEW .
		?actPar osph:hasParamSimpleValue ?actParSimpValNEW .
		?actPar osph:hasParamVariable ?actParVarNEW .
		?actPar osph:hasEventPayloadVariable ?actParEvVarNEW .
		?actPar osph:hasEventPayloadValue ?actParEvValNEW .
	}
}
WHERE{

	#?statemach rdf:type fsm:StateMachine .
	?action rdf:type/rdfs:subClassOf* fsm:Action .

	# VALUES (?actParName ) {("delay" ) } 
	
	OPTIONAL { 
		FILTER(BOUND(?actParName))
		GRAPH ?g0{
			?action osph:hasActionParameter ?actParOLD .
			?actParOLD osph:hasParamName ?actParNameOLD .
			#?actParOLD ?propOLD ?objOLD .
			OPTIONAL{?actParOLD osph:hasParamValue ?actParValOLD .}
			OPTIONAL{?actParOLD osph:hasParamSimpleValue ?actParSimpValOLD .}
			OPTIONAL{?actParOLD osph:hasParamVariable ?actParVarOLD .}
			OPTIONAL{?actParOLD osph:hasEventPayloadVariable ?actParEvVarOLD .}
			OPTIONAL{?actParOLD osph:hasEventPayloadValue ?actParEvValOLD .}	
		}
		FILTER(?actParNameOLD = ?actParName)
	}	

	# BIND(IF(bound(?actParName) , URI(CONCAT(STR(?action),"_par",STR(?actParName))) , ?null) AS ?actPar ) .
	BIND(IF(BOUND(?actParName),IF(BOUND(?actParOLD),?actParOLD,URI(CONCAT(STR(?action),"_par_",STR(?actParName)))),?null) AS ?actPar) .

	BIND(IF(bound(?actParVal) , ?actParValOLD , ?null) AS ?actParValDEL ) .	
	BIND(IF(bound(?actParSimpVal) || bound(?actParVar) , ?actParSimpValOLD , ?null) AS ?actParSimpValDEL ) .	
	BIND(IF(bound(?actParVar) || bound(?actParSimpVal) , ?actParVarOLD , ?null) AS ?actParVarDEL ) .	
	BIND(IF(bound(?actParEvVar) , ?actParEvVarOLD , ?null) AS ?actParEvVarDEL ) .	
	# BIND(IF(bound(?actParEvVal) , ?actParEvValOLD , ?null) AS ?actParEvValDEL ) .	
	BIND(IF(bound(?actParEvVal) || bound(?actParEvVar) , ?actParEvValOLD , ?null) AS ?actParEvValDEL ) .	
	
	BIND(IF(bound(?actParVal) && !CONTAINS(STR(?actParVal),"*DELETE*"), ?actParVal , ?null) AS ?actParValNEW ) .	
	BIND(IF(bound(?actParSimpVal) , ?actParSimpVal , ?null) AS ?actParSimpValNEW ) .	
	BIND(IF(bound(?actParVar) && !CONTAINS(STR(?actParVar),"*DELETE*") , ?actParVar , ?null) AS ?actParVarNEW ) .	
	BIND(IF(bound(?actParEvVar) && !CONTAINS(STR(?actParEvVar),"*DELETE*") , ?actParEvVar , ?null) AS ?actParEvVarNEW ) .	
	BIND(IF(bound(?actParEvVal) , ?actParEvVal , ?null) AS ?actParEvValNEW ) .	
}
